function deleteClientDocs(orderId) {
const ord = state.orders.find(o => o.id === orderId);
if (ord) {
ord.hasDocs = false;
alert('¡Tus documentos de cédula y selfie han sido eliminados permanentemente del servidor con éxito!');
renderApp();
}
}

function updateOrderStatus(id, status) {
const ord = state.orders.find(o => o.id === id);
if (ord) { ord.status = status; renderApp(); }
}

function addToCart(id) {
const prod = state.products.find(p => p.id === id);
const existing = state.cart.find(i => i.id === id);
if (existing) existing.qty++;
else state.cart.push({ ...prod, qty: 1 });
updateCartUI();
toggleCartModal(true);
}

function updateCartUI() {
document.getElementById('cart-count').innerText = state.cart.reduce((a, b) => a + b.qty, 0);
const subtotal = state.cart.reduce((a, b) => a + (b.price * b.qty), 0);
const total = subtotal > 0 ? subtotal + 0.50 : 0;

document.getElementById('cart-subtotal').innerText = `$${subtotal.toFixed(2)}`;
document.getElementById('cart-total').innerText = `$${total.toFixed(2)}`;

const container = document.getElementById('cart-items-container');
if (state.cart.length === 0) {
container.innerHTML = `<p class="text-center text-slate-400 py-12 text-xs">Tu carrito está vacío.</p>`;
return;
}
container.innerHTML = state.cart.map(i => `
<div class="flex justify-between items-center border-b pb-2 text-xs">
<span>${i.qty}x ${i.name}</span>
<span class="font-bold text-emerald-600">$${(i.price * i.qty).toFixed(2)}</span>
</div>
`).join('');
}

function toggleCartModal(show) {
const modal = document.getElementById('cart-modal');
if (show) modal.classList.remove('hidden');
else modal.classList.add('hidden');
}

function toggleAuthModal(show) {
const modal = document.getElementById('auth-modal');
if (show) modal.classList.remove('hidden');
else modal.classList.add('hidden');
}

function togglePaymentFields(method) {
const pmFields = document.getElementById('pago-movil-fields');
const creditFields = document.getElementById('credit-verification-fields');
if (method === 'pago_movil') { pmFields.classList.remove('hidden'); creditFields.classList.add('hidden'); }
else if (method === 'credit') { pmFields.classList.add('hidden'); creditFields.classList.remove('hidden'); }
else { pmFields.classList.add('hidden'); creditFields.classList.add('hidden'); }
}

function handleRegister() {
const name = document.getElementById('reg-name').value;
const cedula = document.getElementById('reg-cedula').value;
if (!name || !cedula) { alert('Completa los campos obligatorios.'); return; }
state.user.name = name;
state.user.cedula = cedula;
alert(`¡Bienvenido, ${name}!`);
toggleAuthModal(false);
document.getElementById('auth-section').innerHTML = `<span class="text-xs font-bold">${name}</span>`;
}

function checkoutViaWhatsApp() {
if (state.cart.length === 0) { alert('Carrito vacío.'); return; }
const paymentMethod = document.getElementById('payment-method').value;

if (paymentMethod === 'credit') {
const cedulaFile = document.getElementById('credit-cedula-file').files[0];
const selfieFile = document.getElementById('credit-selfie-file').files[0];
if (!cedulaFile || !selfieFile) {
alert('Para compras a crédito debes adjuntar la foto de tu cédula y tu selfie.');
return;
}
}

const subtotal = state.cart.reduce((acc, item) => acc + (item.price * item.qty), 0);
const total = subtotal + 0.50;

let message = `🛒 *COMPRA FÁCIL* 🇻🇪\nCliente: ${state.user.name}\nMétodo: ${paymentMethod}\nTotal: $${total.toFixed(2)}`;
window.open(`https://wa.me/584120000000?text=${encodeURIComponent(message)}`, '_blank');
state.cart = [];
toggleCartModal(false);
router('tracking');
}

renderApp();
